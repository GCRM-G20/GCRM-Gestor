   146→function ParticleBackground() {
   147→  const canvasRef = useRef<HTMLCanvasElement>(null);
   148→
   149→  useEffect(() => {
   150→    const canvas = canvasRef.current;
   151→    if (!canvas) return;
   152→    const ctx = canvas.getContext('2d');
   153→    if (!ctx) return;
   154→    let animId: number;
   155→
   156→    const resize = () => {
   157→      canvas.width = window.innerWidth;
   158→      canvas.height = window.innerHeight;
   159→    };
   160→    resize();
   161→    window.addEventListener('resize', resize);
   162→
   163→    const PARTICLE_COUNT = 80;
   164→    const CONNECTION_DIST = 150;
   165→    const particles: { x: number; y: number; vx: number; vy: number; r: number; o: number }[] = [];
   166→
   167→    for (let i = 0; i < PARTICLE_COUNT; i++) {
   168→      particles.push({
   169→        x: Math.random() * canvas.width,
   170→        y: Math.random() * canvas.height,
   171→        vx: (Math.random() - 0.5) * 0.4,
   172→        vy: (Math.random() - 0.5) * 0.4,
   173→        r: Math.random() * 1.5 + 0.5,
   174→        o: Math.random() * 0.5 + 0.2,
   175→      });
   176→    }
   177→
   178→    const draw = () => {
   179→      ctx.clearRect(0, 0, canvas.width, canvas.height);
   180→
   181→      // Update & draw particles
   182→      for (const p of particles) {
   183→        p.x += p.vx;
   184→        p.y += p.vy;
   185→        if (p.x < 0 || p.x > canvas.width) p.vx *= -1;
   186→        if (p.y < 0 || p.y > canvas.height) p.vy *= -1;
   187→
   188→        ctx.beginPath();
   189→        ctx.arc(p.x, p.y, p.r, 0, Math.PI * 2);
   190→        ctx.fillStyle = `rgba(240, 185, 11, ${p.o})`;
   191→        ctx.fill();
   192→      }
   193→
   194→      // Draw connections
   195→      for (let i = 0; i < particles.length; i++) {
   196→        for (let j = i + 1; j < particles.length; j++) {
   197→          const dx = particles[i].x - particles[j].x;
   198→          const dy = particles[i].y - particles[j].y;
   199→          const dist = Math.sqrt(dx * dx + dy * dy);
   200→          if (dist < CONNECTION_DIST) {
   201→            const alpha = (1 - dist / CONNECTION_DIST) * 0.15;
   202→            ctx.beginPath();
   203→            ctx.moveTo(particles[i].x, particles[i].y);
   204→            ctx.lineTo(particles[j].x, particles[j].y);
   205→            ctx.strokeStyle = `rgba(240, 185, 11, ${alpha})`;
   206→            ctx.lineWidth = 0.5;
   207→            ctx.stroke();
   208→          }
   209→        }
   210→      }
   211→
   212→      animId = requestAnimationFrame(draw);
   213→    };
   214→    draw();
   215→
   216→    return () => {
   217→      cancelAnimationFrame(animId);
   218→      window.removeEventListener('resize', resize);
   219→    };
   220→  }, []);
   221→
   222→  return (
   223→    <canvas
   224→      ref={canvasRef}
   225→      className="absolute inset-0 w-full h-full pointer-events-none"
   226→      style={{ zIndex: 1 }}
   227→    />
   228→  );
   229→}

   233→function ExchangeCarousel() {
   234→  const scrollRef = useRef<HTMLDivElement>(null);
   235→  const [canScrollLeft, setCanScrollLeft] = useState(false);
   236→  const [canScrollRight, setCanScrollRight] = useState(true);
   237→
   238→  const checkScroll = useCallback(() => {
   239→    const el = scrollRef.current;
   240→    if (!el) return;
   241→    setCanScrollLeft(el.scrollLeft > 0);
   242→    setCanScrollRight(el.scrollLeft < el.scrollWidth - el.clientWidth - 10);
   243→  }, []);
   244→
   245→  useEffect(() => {
   246→    checkScroll();
   247→    const el = scrollRef.current;
   248→    if (el) el.addEventListener('scroll', checkScroll);
   249→    return () => { if (el) el.removeEventListener('scroll', checkScroll); };
   250→  }, [checkScroll]);
   251→
   252→  const scroll = (dir: 'left' | 'right') => {
   253→    const el = scrollRef.current;
   254→    if (!el) return;
   255→    el.scrollBy({ left: dir === 'left' ? -200 : 200, behavior: 'smooth' });
   256→  };
   257→
   258→  const items = [...EXCHANGES, ...EXCHANGES];
   259→
   260→  return (
   261→    <div className="relative w-full">
   262→      <div className="text-center mb-6">
   263→        <p className="text-[#848E9C] text-sm uppercase tracking-widest mb-1">Listado en Exchanges</p>

   269→        </p>
   270→        <p className="text-[#F0B90B] text-sm font-semibold mt-2">Próximamente: GCRM Exchange + GCRM Wallet.</p>
   271→      </div>
   272→      <div className="relative group">
   273→        {canScrollLeft && (
   274→          <button onClick={() => scroll('left')} className="absolute left-0 top-1/2 -translate-y-1/2 z-10 w-10 h-10 rounded-full bg-[#1E2329] border border-[#2B3139] flex items-center justify-center hover:bg-[#2B3139] transition-all opacity-0 group-hover:opacity-100" aria-label="Anterior">
   275→            <ChevronLeft className="w-5 h-5 text-[#EAECEF]" />

   279→          <button onClick={() => scroll('right')} className="absolute right-0 top-1/2 -translate-y-1/2 z-10 w-10 h-10 rounded-full bg-[#1E2329] border border-[#2B3139] flex items-center justify-center hover:bg-[#2B3139] transition-all opacity-0 group-hover:opacity-100" aria-label="Siguiente">
   280→            <ChevronRight className="w-5 h-5 text-[#EAECEF]" />
   281→          </button>
   282→        )}
   283→        <div ref={scrollRef} className="flex gap-6 overflow-x-auto py-4 px-2 snap-x snap-mandatory" style={{ scrollbarWidth: 'none', msOverflowStyle: 'none' }}>
   284→          {items.map((ex, i) => (
   285→            <div key={`${ex.name}-${i}`} className="snap-center shrink-0">
   286→              <ExchangeLogo {...ex} />
   287→            </div>

   294→        <div className="h-px bg-[#2B3139] w-16" />
   295→      </div>
   296→    </div>
   297→  );

   345→function LoginDialog({ open, onOpenChange, onSwitchToRegister }: {

   348→  onSwitchToRegister: () => void;
   349→}) {

   353→  const [loading, setLoading] = useState(false);
   354→  const { toast } = useToast();

   356→  const handleLogin = async (e: React.FormEvent) => {
   357→    e.preventDefault();

   359→      toast({ title: 'Campos requeridos', description: 'Ingresa correo y contraseña.', variant: 'destructive' });

   383→

   429→function RegisterDialog({ open, onOpenChange, onSwitchToLogin }: {
   430→  open: boolean;
   431→  onOpenChange: (open: boolean) => void;
   432→  onSwitchToLogin: () => void;
   433→}) {

   440→  const { toast } = useToast();

   442→  const handleRegister = async (e: React.FormEvent) => {
   443→    e.preventDefault();

   445→      toast({ title: 'Campos requeridos', description: 'Completa todos los campos.', variant: 'destructive' });
   446→      return;
   447→    }
   448→    if (password.length < 6) {

   456→    setLoading(true);
   457→    try {
   458→      const res = await fetch('/api/auth/register', {
   459→        method: 'POST',
   460→        headers: { 'Content-Type': 'application/json' },
   461→        body: JSON.stringify({ name, email, password, licencia: regLicencia }),
   462→      });
   463→      const data = await res.json();
   464→      if (!res.ok) {
   465→        toast({ title: 'Error', description: data.error, variant: 'destructive' });
   466→      } else {
   467→        toast({ title: 'Registro exitoso', description: `Bienvenido/a ${data.user.name}. Licencia: ${regLicencia ? LICENCIAS.find(l => l.id === regLicencia)?.label : 'N/A'}.` });
   468→        onOpenChange(false);
   469→        setName(''); setEmail(''); setPassword('');
