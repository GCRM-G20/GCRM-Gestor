   144→   144→/* ─── PARTICLE BACKGROUND ─── */
   145→   145→
   146→   146→function ParticleBackground() {
   147→   147→  const canvasRef = useRef<HTMLCanvasElement>(null);
   148→   148→
   149→   149→  useEffect(() => {
   150→   150→    const canvas = canvasRef.current;
   151→   151→    if (!canvas) return;
   152→   152→    const ctx = canvas.getContext('2d');
   153→   153→    if (!ctx) return;
   154→   154→    let animId: number;
   155→   155→
   156→   156→    const resize = () => {
   157→   157→      canvas.width = window.innerWidth;
   158→   158→      canvas.height = window.innerHeight;
   159→   159→    };
   160→   160→    resize();
   161→   161→    window.addEventListener('resize', resize);
   162→   162→
   163→   163→    const PARTICLE_COUNT = 80;
   164→   164→    const CONNECTION_DIST = 150;
   165→   165→    const particles: { x: number; y: number; vx: number; vy: number; r: number; o: number }[] = [];
   166→   166→
   167→   167→    for (let i = 0; i < PARTICLE_COUNT; i++) {
   168→   168→      particles.push({
   169→   169→        x: Math.random() * canvas.width,
   170→   170→        y: Math.random() * canvas.height,
   171→   171→        vx: (Math.random() - 0.5) * 0.4,
   172→   172→        vy: (Math.random() - 0.5) * 0.4,
   173→   173→        r: Math.random() * 1.5 + 0.5,
   174→   174→        o: Math.random() * 0.5 + 0.2,
   175→   175→      });
   176→   176→    }
   177→   177→
   178→   178→    const draw = () => {
   179→   179→      ctx.clearRect(0, 0, canvas.width, canvas.height);
   180→   180→
   181→   181→      // Update & draw particles
   182→   182→      for (const p of particles) {
   183→   183→        p.x += p.vx;
   184→   184→        p.y += p.vy;
   185→   185→        if (p.x < 0 || p.x > canvas.width) p.vx *= -1;
   186→   186→        if (p.y < 0 || p.y > canvas.height) p.vy *= -1;
   187→   187→
   188→   188→        ctx.beginPath();
   189→   189→        ctx.arc(p.x, p.y, p.r, 0, Math.PI * 2);
   190→   190→        ctx.fillStyle = `rgba(240, 185, 11, ${p.o})`;
   191→   191→        ctx.fill();
   192→   192→      }
   193→   193→
   194→   194→      // Draw connections
   195→   195→      for (let i = 0; i < particles.length; i++) {
   196→   196→        for (let j = i + 1; j < particles.length; j++) {
   197→   197→          const dx = particles[i].x - particles[j].x;
   198→   198→          const dy = particles[i].y - particles[j].y;
   199→   199→          const dist = Math.sqrt(dx * dx + dy * dy);
   200→   200→          if (dist < CONNECTION_DIST) {
   201→   201→            const alpha = (1 - dist / CONNECTION_DIST) * 0.15;
   202→   202→            ctx.beginPath();
   203→   203→            ctx.moveTo(particles[i].x, particles[i].y);
   204→   204→            ctx.lineTo(particles[j].x, particles[j].y);
   205→   205→            ctx.strokeStyle = `rgba(240, 185, 11, ${alpha})`;
   206→   206→            ctx.lineWidth = 0.5;
   207→   207→            ctx.stroke();
   208→   208→          }
   209→   209→        }
   210→   210→      }
   211→   211→
   212→   212→      animId = requestAnimationFrame(draw);
   213→   213→    };
   214→   214→    draw();
   215→   215→
   216→   216→    return () => {
   217→   217→      cancelAnimationFrame(animId);
   218→   218→      window.removeEventListener('resize', resize);
   219→   219→    };
   220→   220→  }, []);
   221→   221→
   222→   222→  return (
   223→   223→    <canvas
   224→   224→      ref={canvasRef}
   225→   225→      className="absolute inset-0 w-full h-full pointer-events-none"
   226→   226→      style={{ zIndex: 1 }}
   227→   227→    />
   228→   228→  );
   229→   229→}

   233→   233→function ExchangeCarousel() {

   235→   235→  const [canScrollLeft, setCanScrollLeft] = useState(false);
   236→   236→  const [canScrollRight, setCanScrollRight] = useState(true);
   237→   237→
   238→   238→  const checkScroll = useCallback(() => {
   239→   239→    const el = scrollRef.current;
   240→   240→    if (!el) return;
   241→   241→    setCanScrollLeft(el.scrollLeft > 0);
   242→   242→    setCanScrollRight(el.scrollLeft < el.scrollWidth - el.clientWidth - 10);
   243→   243→  }, []);
   244→   244→
   245→   245→  useEffect(() => {
   246→   246→    checkScroll();
   247→   247→    const el = scrollRef.current;
   248→   248→    if (el) el.addEventListener('scroll', checkScroll);
   249→   249→    return () => { if (el) el.removeEventListener('scroll', checkScroll); };
   250→   250→  }, [checkScroll]);
   251→   251→
   252→   252→  const scroll = (dir: 'left' | 'right') => {
   253→   253→    const el = scrollRef.current;
   254→   254→    if (!el) return;
   255→   255→    el.scrollBy({ left: dir === 'left' ? -200 : 200, behavior: 'smooth' });
   256→   256→  };

   271→   271→      </div>
   272→   272→      <div className="relative group">
   273→   273→        {canScrollLeft && (
   274→   274→          <button onClick={() => scroll('left')} className="absolute left-0 top-1/2 -translate-y-1/2 z-10 w-10 h-10 rounded-full bg-[#1E2329] border border-[#2B3139] flex items-center justify-center hover:bg-[#2B3139] transition-all opacity-0 group-hover:opacity-100" aria-label="Anterior">

   280→   280→            <ChevronRight className="w-5 h-5 text-[#EAECEF]" />
