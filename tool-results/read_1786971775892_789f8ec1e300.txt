   100→function CopyButton({ text }: { text: string }) {
   101→  const [copied, setCopied] = useState(false);
   102→  const { toast } = useToast();
   103→
   104→  const handleCopy = async () => {
   105→    try {
   106→      await navigator.clipboard.writeText(text);
   107→      setCopied(true);
   108→      toast({ title: 'Dirección copiada', description: 'La dirección ha sido copiada al portapapeles.' });

   111→      toast({ title: 'Error al copiar', description: 'No se pudo copiar la dirección.', variant: 'destructive' });

   113→  };
   114→
   115→  return (
   116→    <button onClick={handleCopy} className="shrink-0 p-2 rounded-lg bg-[#2B3139] hover:bg-[#3B434D] transition-colors" aria-label="Copiar dirección">
   117→      {copied ? <Check className="w-4 h-4 text-[#0ECB81]" /> : <Copy className="w-4 h-4 text-[#848E9C]" />}

   136→/* ─── EXCHANGE CAROUSEL ─── */
   137→
   138→function ExchangeCarousel() {
   139→  const scrollRef = useRef<HTMLDivElement>(null);
   140→  const [canScrollLeft, setCanScrollLeft] = useState(false);
   141→  const [canScrollRight, setCanScrollRight] = useState(true);
   142→
   143→  const checkScroll = useCallback(() => {
   144→    const el = scrollRef.current;
   145→    if (!el) return;
   146→    setCanScrollLeft(el.scrollLeft > 0);
   147→    setCanScrollRight(el.scrollLeft < el.scrollWidth - el.clientWidth - 10);
   148→  }, []);
   149→
   150→  useEffect(() => {
   151→    checkScroll();
   152→    const el = scrollRef.current;
   153→    if (el) el.addEventListener('scroll', checkScroll);
   154→    return () => { if (el) el.removeEventListener('scroll', checkScroll); };
   155→  }, [checkScroll]);
   156→
   157→  const scroll = (dir: 'left' | 'right') => {
   158→    const el = scrollRef.current;
   159→    if (!el) return;
   160→    el.scrollBy({ left: dir === 'left' ? -200 : 200, behavior: 'smooth' });
   161→  };
   162→
   163→  const items = [...EXCHANGES, ...EXCHANGES];
   164→
   165→  return (
   166→    <div className="relative w-full">
   167→      <div className="text-center mb-6">
   168→        <p className="text-[#848E9C] text-sm uppercase tracking-widest mb-1">Listado en Exchanges</p>
   169→        <h2 className="text-xl md:text-2xl font-bold text-[#EAECEF]">
   170→          Disponible en los Mejores <span className="text-gold-gradient">Exchanges</span> del Mundo
   171→        </h2>
   172→      </div>
   173→      <div className="relative group">
   174→        {canScrollLeft && (
   175→          <button onClick={() => scroll('left')} className="absolute left-0 top-1/2 -translate-y-1/2 z-10 w-10 h-10 rounded-full bg-[#1E2329] border border-[#2B3139] flex items-center justify-center hover:bg-[#2B3139] transition-all opacity-0 group-hover:opacity-100" aria-label="Anterior">
   176→            <ChevronLeft className="w-5 h-5 text-[#EAECEF]" />
   177→          </button>
   178→        )}
   179→        {canScrollRight && (
   180→          <button onClick={() => scroll('right')} className="absolute right-0 top-1/2 -translate-y-1/2 z-10 w-10 h-10 rounded-full bg-[#1E2329] border border-[#2B3139] flex items-center justify-center hover:bg-[#2B3139] transition-all opacity-0 group-hover:opacity-100" aria-label="Siguiente">
   181→            <ChevronRight className="w-5 h-5 text-[#EAECEF]" />
   182→          </button>
   183→        )}
   184→        <div ref={scrollRef} className="flex gap-6 overflow-x-auto py-4 px-2 snap-x snap-mandatory" style={{ scrollbarWidth: 'none', msOverflowStyle: 'none' }}>
   185→          {items.map((ex, i) => (
   186→            <div key={`${ex.name}-${i}`} className="snap-center shrink-0">
   187→              <ExchangeLogo {...ex} />
   188→            </div>
   189→          ))}
   190→        </div>
   191→      </div>
   192→      <div className="flex items-center justify-center gap-2 mt-4">
   193→        <div className="h-px bg-[#2B3139] w-16" />
   194→        <span className="text-[#848E9C] text-xs uppercase tracking-wider">Verificado en CoinMarketCap</span>
   195→        <div className="h-px bg-[#2B3139] w-16" />
   196→      </div>
   197→    </div>
   198→  );
   199→}

   245→
   246→function LoginDialog({ open, onOpenChange, onSwitchToRegister }: {
   247→  open: boolean;
   248→  onOpenChange: (open: boolean) => void;
   249→  onSwitchToRegister: () => void;
   250→}) {
   251→  const [email, setEmail] = useState('');
   252→  const [password, setPassword] = useState('');
   253→  const [showPassword, setShowPassword] = useState(false);
   254→  const [loading, setLoading] = useState(false);
   255→  const { toast } = useToast();
   256→
   257→  const handleLogin = async (e: React.FormEvent) => {
   258→    e.preventDefault();
   259→    if (!email || !password) {
   260→      toast({ title: 'Campos requeridos', description: 'Ingresa correo y contraseña.', variant: 'destructive' });
   261→      return;
   262→    }
   263→    setLoading(true);
   264→    try {
   265→      const res = await fetch('/api/auth/login', {
   266→        method: 'POST',
   267→        headers: { 'Content-Type': 'application/json' },
   268→        body: JSON.stringify({ email, password }),
   269→      });
   270→      const data = await res.json();
   271→      if (!res.ok) {
   272→        toast({ title: 'Error', description: data.error, variant: 'destructive' });
   273→      } else {
   274→        toast({ title: 'Bienvenido', description: `Hola, ${data.user.name}.` });
   275→        onOpenChange(false);
   276→        setEmail(''); setPassword('');
   277→        window.location.reload();
   278→      }
   279→    } catch {
   280→      toast({ title: 'Error', description: 'No se pudo conectar al servidor.', variant: 'destructive' });
   281→    }
   282→    setLoading(false);
   283→  };
   284→
   285→  return (
   286→    <Dialog open={open} onOpenChange={onOpenChange}>
   287→      <DialogContent className="bg-[#1E2329] border-[#2B3139] text-[#EAECEF] sm:max-w-md">
   288→        <DialogHeader>
   289→          <DialogTitle className="text-xl font-bold flex items-center gap-2">
   290→            <LogIn className="w-5 h-5 text-[#F0B90B]" />
   291→            Iniciar Sesión
   292→          </DialogTitle>
   293→          <DialogDescription className="text-[#848E9C]">
   294→            Ingresa tus credenciales para acceder a tu panel de comisiones.
   295→          </DialogDescription>
   296→        </DialogHeader>
   297→        <form onSubmit={handleLogin} className="space-y-4 mt-4">
   298→          <div className="space-y-2">
   299→            <Label htmlFor="login-email" className="text-sm text-[#EAECEF]">Correo Electrónico</Label>
   300→            <Input id="login-email" type="email" placeholder="tu@correo.com" value={email} onChange={(e) => setEmail(e.target.value)}
   301→              className="bg-[#2B3139] border-[#2B3139] text-[#EAECEF] placeholder:text-[#5E6673] focus:border-[#F0B90B] focus:ring-[#F0B90B]/20 rounded-xl h-11 text-sm" />
   302→          </div>
   303→          <div className="space-y-2">
   304→            <Label htmlFor="login-pass" className="text-sm text-[#EAECEF]">Contraseña</Label>
   305→            <div className="relative">
   306→              <Input id="login-pass" type={showPassword ? 'text' : 'password'} placeholder="Tu contraseña" value={password} onChange={(e) => setPassword(e.target.value)}
   307→                className="bg-[#2B3139] border-[#2B3139] text-[#EAECEF] placeholder:text-[#5E6673] focus:border-[#F0B90B] focus:ring-[#F0B90B]/20 rounded-xl h-11 text-sm pr-10" />
   308→              <button type="button" onClick={() => setShowPassword(!showPassword)} className="absolute right-3 top-1/2 -translate-y-1/2 text-[#848E9C] hover:text-[#EAECEF] transition-colors">
   309→                {showPassword ? <EyeOff className="w-4 h-4" /> : <Eye className="w-4 h-4" />}
   310→              </button>
   311→            </div>
   312→          </div>
   313→          <Button type="submit" disabled={loading} className="w-full bg-[#F0B90B] hover:bg-[#F8D12F] text-[#0B0E11] font-bold rounded-xl h-11 text-sm transition-all disabled:opacity-50">
   314→            {loading ? <Loader2 className="w-4 h-4 animate-spin" /> : 'Iniciar Sesión'}
   315→          </Button>
   316→          <p className="text-center text-sm text-[#848E9C]">
   317→            ¿No tienes cuenta?{' '}
   318→            <button type="button" onClick={() => { onOpenChange(false); onSwitchToRegister(); }} className="text-[#F0B90B] hover:underline font-medium">
   319→              Regístrate aquí
   320→            </button>
   321→          </p>
   322→        </form>
   323→      </DialogContent>
   324→    </Dialog>
   325→  );
   326→}

   330→function RegisterDialog({ open, onOpenChange, onSwitchToLogin }: {

   340→  const [loading, setLoading] = useState(false);
   341→  const { toast } = useToast();
   342→
   343→  const handleRegister = async (e: React.FormEvent) => {
   344→    e.preventDefault();
   345→    if (!name || !email || !password) {
   346→      toast({ title: 'Campos requeridos', description: 'Completa todos los campos.', variant: 'destructive' });
