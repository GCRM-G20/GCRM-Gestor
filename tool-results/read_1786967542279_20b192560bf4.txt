    75→function CopyButton({ text }: { text: string }) {
    76→  const [copied, setCopied] = useState(false);
    77→  const { toast } = useToast();
    78→
    79→  const handleCopy = async () => {
    80→    try {
    81→      await navigator.clipboard.writeText(text);
    82→      setCopied(true);
    83→      toast({
    84→        title: 'Dirección copiada',
    85→        description: 'La dirección ha sido copiada al portapapeles.',
    86→      });
    87→      setTimeout(() => setCopied(false), 2000);
    88→    } catch {
    89→      toast({
    90→        title: 'Error al copiar',
    91→        description: 'No se pudo copiar la dirección.',
    92→        variant: 'destructive',
    93→      });
    94→    }
    95→  };
    96→
    97→  return (
    98→    <button
    99→      onClick={handleCopy}
   100→      className="shrink-0 p-2 rounded-lg bg-[#2B3139] hover:bg-[#3B434D] transition-colors"
   101→      aria-label="Copiar dirección"

   103→      {copied ? <Check className="w-4 h-4 text-green-400" /> : <Copy className="w-4 h-4 text-[#848E9C]" />}

   126→/* ─── EXCHANGE CAROUSEL ─── */
   127→
   128→function ExchangeCarousel() {
   129→  const scrollRef = useRef<HTMLDivElement>(null);
   130→  const [canScrollLeft, setCanScrollLeft] = useState(false);
   131→  const [canScrollRight, setCanScrollRight] = useState(true);
   132→
   133→  const checkScroll = useCallback(() => {
   134→    const el = scrollRef.current;
   135→    if (!el) return;
   136→    setCanScrollLeft(el.scrollLeft > 0);
   137→    setCanScrollRight(el.scrollLeft < el.scrollWidth - el.clientWidth - 10);
   138→  }, []);
   139→
   140→  useEffect(() => {
   141→    checkScroll();
   142→    const el = scrollRef.current;
   143→    if (el) el.addEventListener('scroll', checkScroll);
   144→    return () => { if (el) el.removeEventListener('scroll', checkScroll); };
   145→  }, [checkScroll]);
   146→
   147→  const scroll = (dir: 'left' | 'right') => {
   148→    const el = scrollRef.current;
   149→    if (!el) return;
   150→    const amount = 200;
   151→    el.scrollBy({ left: dir === 'left' ? -amount : amount, behavior: 'smooth' });
   152→  };
   153→
   154→  // Double the items for infinite feel
   155→  const items = [...EXCHANGES, ...EXCHANGES];
   156→
   157→  return (
   158→    <div className="relative w-full">
   159→      {/* CoinMarketCap style header */}
   160→      <div className="text-center mb-6">
   161→        <p className="text-[#848E9C] text-sm uppercase tracking-widest mb-1">Listado en Exchanges</p>
   162→        <h2 className="text-xl md:text-2xl font-bold text-[#EAECEF]">
   163→          Disponible en los Mejores <span className="text-gold-gradient">Exchanges</span> del Mundo
   164→        </h2>
   165→      </div>
   166→
   167→      {/* Scrollable container */}
   168→      <div className="relative group">
   169→        {canScrollLeft && (
   170→          <button
   171→            onClick={() => scroll('left')}
   172→            className="absolute left-0 top-1/2 -translate-y-1/2 z-10 w-10 h-10 rounded-full bg-[#1E2329] border border-[#2B3139] flex items-center justify-center hover:bg-[#2B3139] transition-all opacity-0 group-hover:opacity-100"
   173→            aria-label="Anterior"
   174→          >
   175→            <ChevronUp className="w-5 h-5 rotate-[-90deg] text-[#EAECEF]" />
   176→          </button>
   177→        )}
   178→        {canScrollRight && (
   179→          <button
   180→            onClick={() => scroll('right')}
   181→            className="absolute right-0 top-1/2 -translate-y-1/2 z-10 w-10 h-10 rounded-full bg-[#1E2329] border border-[#2B3139] flex items-center justify-center hover:bg-[#2B3139] transition-all opacity-0 group-hover:opacity-100"
   182→            aria-label="Siguiente"
   183→          >
   184→            <ChevronDown className="w-5 h-5 rotate-90 text-[#EAECEF]" />
   185→          </button>
   186→        )}
   187→
   188→        <div
   189→          ref={scrollRef}
   190→          className="flex gap-6 overflow-x-auto py-4 px-2 snap-x snap-mandatory scrollbar-hide"
   191→          style={{ scrollbarWidth: 'none', msOverflowStyle: 'none' }}
   192→        >
   193→          {items.map((ex, i) => (
   194→            <div key={`${ex.name}-${i}`} className="snap-center shrink-0">
   195→              <ExchangeLogo {...ex} />
   196→            </div>
   197→          ))}
   198→        </div>
   199→      </div>
   200→
   201→      {/* CoinMarketCap badge */}
   202→      <div className="flex items-center justify-center gap-2 mt-4">
   203→        <div className="h-px bg-[#2B3139] w-16" />
   204→        <span className="text-[#848E9C] text-xs uppercase tracking-wider">Verificado en CoinMarketCap</span>
   205→        <div className="h-px bg-[#2B3139] w-16" />
   206→      </div>
   207→    </div>
   208→  );
   209→}
